# ==================== 1. 城市区划地块 ====================
mod_extend_urban_workplace = {
    icon = "d_monument"
    is_for_colonizable = yes
    potential = { always = no }
    should_swap_deposit_on_terraforming = no

    planet_modifier = {
        planet_housing_add = 1200
        planet_max_districts_add = 2
        planet_amenities_add = 1200
        planet_building_build_speed_mult = 0.20
    }

    # 非格式塔帝国：增加政治家
    triggered_planet_modifier = {
        potential = { exists = owner owner = { is_gestalt = no } }
        modifier = {
            job_politician_add = 200
            pop_cat_ruler_bonus_workforce_mult = 0.1
        }
    }

    drop_weight = { weight = 0 }
}
