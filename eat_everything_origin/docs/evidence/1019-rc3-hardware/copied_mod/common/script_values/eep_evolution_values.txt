# Ship scope: each installed core reads its own fleet's persistent rank.
# An absent fleet or unset rank contributes nothing; no artificial rank cap.
eep_component_shield_level_value = {
    base = 0
    modifier = {
        exists = fleet
        fleet = { is_variable_set = eep_component_shield_level }
        add = fleet.eep_component_shield_level
    }
}
