country_event = {
    id = eep.2
    hide_window = yes
    is_triggered_only = yes
    trigger = { has_origin = origin_heart_of_devouring }
    immediate = {
        if = {
            limit = { has_origin = origin_heart_of_devouring }
            eep_initialize = yes
            if = {
                limit = {
                    has_country_flag = eep_initialized
                    NOT = { has_country_flag = eep_core_dead }
                    OR = {
                        NOT = { exists = event_target:eep_core@this }
                        event_target:eep_core@this = { is_colony = no }
                    }
                }
                set_country_flag = eep_core_dead
                country_event = { id = eep.42 }
            }
            else_if = {
                limit = { NOT = { has_country_flag = eep_core_dead } }
                if = {
                    limit = { eep_core_available = no NOT = { has_country_flag = eep_core_lost } }
                    set_country_flag = eep_core_lost
                    country_event = { id = eep.40 }
                }
                else_if = {
                    limit = { eep_core_available = yes has_country_flag = eep_core_lost }
                    remove_country_flag = eep_core_lost
                    country_event = { id = eep.41 }
                }
                if = {
                    limit = { has_country_flag = eep_notice_pending }
                    remove_country_flag = eep_notice_pending
                    eep_report = yes
                    if = {
                        limit = { NOT = { has_country_flag = eep_first_notice } }
                        set_country_flag = eep_first_notice
                        country_event = { id = eep.11 }
                    }
                    else_if = {
                        limit = { check_variable = { which = eep_c value >= 200 } check_variable = { which = eep_stage value < 3 } }
                        set_variable = { which = eep_stage value = 3 }
                        country_event = { id = eep.13 }
                    }
                    else_if = {
                        limit = { check_variable = { which = eep_c value >= 60 } check_variable = { which = eep_stage value < 2 } }
                        set_variable = { which = eep_stage value = 2 }
                        country_event = { id = eep.13 }
                    }
                    else_if = {
                        limit = { check_variable = { which = eep_c value >= 20 } check_variable = { which = eep_stage value < 1 } }
                        set_variable = { which = eep_stage value = 1 }
                        country_event = { id = eep.13 }
                    }
                    # The first destruction also absorbs the highest reached milestone.
                    if = { limit = { check_variable = { which = eep_c value >= 20 } check_variable = { which = eep_stage value < 1 } } set_variable = { which = eep_stage value = 1 } }
                    if = { limit = { check_variable = { which = eep_c value >= 60 } check_variable = { which = eep_stage value < 2 } } set_variable = { which = eep_stage value = 2 } }
                    if = { limit = { check_variable = { which = eep_c value >= 200 } check_variable = { which = eep_stage value < 3 } } set_variable = { which = eep_stage value = 3 } }
                }
                if = {
                    limit = { has_finished_psionic_tradition = yes NOT = { has_country_flag = eep_psi_notice } }
                    set_country_flag = eep_psi_notice
                    country_event = { id = eep.30 }
                }
                if = {
                    limit = {
                        has_ascension_perk = ap_become_the_crisis
                        OR = { has_crisis_level = crisis_level_2 has_crisis_level = crisis_level_3 has_crisis_level = crisis_level_4 has_crisis_level = crisis_level_5 }
                        NOT = { has_country_flag = eep_fleet_notice }
                    }
                    set_country_flag = eep_fleet_notice
                    eep_report = yes
                    set_variable = { which = eep_fleet_stage value = eep_crisis }
                    country_event = { id = eep.31 }
                }
                else_if = {
                    limit = { has_ascension_perk = ap_become_the_crisis has_country_flag = eep_fleet_notice has_crisis_level = crisis_level_4 check_variable = { which = eep_fleet_stage value < 4 } }
                    eep_report = yes
                    set_variable = { which = eep_fleet_stage value = 4 }
                    country_event = { id = eep.32 }
                }
                else_if = {
                    limit = { has_ascension_perk = ap_become_the_crisis has_country_flag = eep_fleet_notice has_crisis_level = crisis_level_3 check_variable = { which = eep_fleet_stage value < 3 } }
                    eep_report = yes
                    set_variable = { which = eep_fleet_stage value = 3 }
                    country_event = { id = eep.32 }
                }
            }
        }
    }
}

situation_event = {
    id = eep.20
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        if = {
            limit = { is_situation_type = situation_eep_devouring eep_task_operational = yes }
            owner = { save_event_target_as = eep_actor }
            situation_event = { id = eep.21 days = 1 }
            if = { limit = { target = { has_planet_flag = eep_pending } } eep_settle = yes }
            else = {
                target = {
                    if = {
                        limit = { has_planet_flag = eep_native NOT = { has_planet_flag = recently_eaten_planet } }
                        eep_native_consume = yes
                    }
                }
            }
        }
    }
}

        }
    }
