# Separate native habitat in a settled foreign empire; initially neutral.
event_target:eep_mothership0.owner = {
    save_global_event_target_as = eep_art_actor
    random_country = {
        limit = {
            is_country_type = default
            is_nomadic = no
            exists = capital_scope
            NOT = { is_same_value = event_target:eep_art_actor }
            NOT = { has_origin = origin_heart_of_devouring }
        }
        save_global_event_target_as = eep_art_enemy
        capital_scope = {
            planet = {
                solar_system = {
                    save_global_event_target_as = eep_art_enemy_system
                    random_system_planet = {
                        limit = {
                            is_star = no
                            is_moon = no
                            is_colony = no
                            is_artificial = no
                            NOT = { has_carrier_flag = has_orbital }
                            NOT = { has_planet_flag = habitat_build_site }
                            NOT = { exists = megastructure }
                        }
                        save_global_event_target_as = eep_art_enemy_anchor
                    }
                }
            }
        }
    }
}
event_target:eep_art_enemy_system = {
    spawn_planet = {
        class = pc_habitat
        location = event_target:eep_art_enemy_anchor
        orbit_location = yes
        orbit_angle_offset = 135
        orbit_distance_offset = 9.899
        size = 6
        has_ring = no
        init_effect = {
            set_planet_entity = { entity = habitat_phase_03_entity graphical_culture = event_target:eep_art_enemy }
            set_surveyed = { surveyed = yes surveyor = event_target:eep_art_actor }
            set_all_comms_surveyed = yes
            clear_blockers = yes
            set_planet_flag = megastructure
            set_planet_flag = habitat
            set_planet_flag = eep_art_enemy_target
            set_carrier_flag = habitat_level_1
            save_global_event_target_as = eep_art_enemy_hab
            event_target:eep_art_enemy_anchor = { set_planet_flag = habitat@PREV }
            create_colony = { owner = event_target:eep_art_enemy species = event_target:eep_art_enemy.owner_main_species }
            create_pop_group = { species = event_target:eep_art_enemy.owner_main_species size = 100 }
            set_name = "ART_ENEMY_HAB"
        }
    }
    spawn_orbital_effect = { TYPE = major HABITAT_OWNER = event_target:eep_art_enemy TARGET_PLANET = event_target:eep_art_enemy_anchor }
    set_star_flag = has_habitat
}
event_target:eep_art_enemy_anchor = { set_carrier_flag = has_orbital }
log = "ART1023_ENEMY_HABITAT_CREATED"
