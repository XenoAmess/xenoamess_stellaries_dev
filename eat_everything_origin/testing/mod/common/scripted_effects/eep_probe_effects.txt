# Controlled test construction only. Production never creates these worlds.
eep_probe_create_source = {
    capital_scope = {
        solar_system = {
            random_system_planet = {
                limit = {
                    is_star = no is_colony = no is_artificial = no
                    NOT = { is_planet_class = pc_gas_giant }
                    NOT = { is_variable_set = eep_q }
                    NOT = { has_planet_flag = eep_probe_constructed }
                }
                change_pc = pc_continental
                set_planet_size = $SIZE$
                clear_deposits = yes
                create_colony = { owner = event_target:eep_probe_country species = event_target:eep_probe_country }
                save_global_event_target_as = $TARGET$
                set_planet_flag = eep_probe_constructed
                event_target:eep_probe_country = {
                    event_target:eep_core@this = {
                        random_owned_pop_group = {
                            limit = { eep_founder_pop = yes pop_group_size >= 100 }
                            resettle_pop_group = { POP_GROUP = this PLANET = event_target:$TARGET$ AMOUNT = 100 }
                        }
                    }
                }
                if = { limit = { always = $START$ } eep_begin = yes }
            }
        }
    }
}
