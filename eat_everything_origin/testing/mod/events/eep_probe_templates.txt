country_event = {
    id = eep_probe.61
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        event_target:eep_probe_source = {
            every_owned_pop_group = {
                limit = { is_same_species = root.owner_main_species }
                modify_species = {
                    species = species
                    add_trait = trait_resilient
                    change_scoped_species = yes
                    effect = { save_global_event_target_as = eep_probe_old_template }
                }
            }
        }
        if = {
            limit = { exists = event_target:eep_probe_foreign exists = event_target:eep_probe_old_template }
            event_target:eep_probe_foreign = {
                capital_scope = {
                    create_pop_group = { species = event_target:eep_probe_old_template size = 100 }
                }
            }
        }
    }
}
