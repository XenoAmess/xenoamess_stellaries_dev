    id = eep_probe.18
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        log = "EEP_MONTH country=[Root.GetName] C=[Root.eep_c] G=[Root.eep_g] D=[Root.eep_d] made=[Root.eep_made] worlds=[Root.eep_worlds]"
        every_situation = {
            limit = { is_situation_type = situation_eep_devouring }
            set_variable = { which = eep_probe_progress value = 0 }
            export_trigger_value_to_variable = { trigger = situation_progress variable = eep_probe_progress }
            target = {
                set_variable = { which = eep_probe_real_progress value = prev.eep_probe_progress }
                planet = { log = "EEP_MONTH source=[This.GetName] Q=[This.eep_q] T=[This.eep_months] progress=[This.eep_probe_real_progress]" }
            }
            if = { limit = { eep_task_operational = yes } log = "EEP_MONTH operational=YES" }
            else = { log = "EEP_MONTH operational=NO" }
            if = { limit = { eep_task_ready = yes } log = "EEP_MONTH ready=YES" }
        }
        if = { limit = { host_has_dlc = "Nemesis" } log = "EEP_DLC Nemesis=YES" }
        else = { log = "EEP_DLC Nemesis=NO" }
        if = { limit = { host_has_dlc = "Shadows of the Shroud" } log = "EEP_DLC Shroud=YES" }
        else = { log = "EEP_DLC Shroud=NO" }
    }
}
