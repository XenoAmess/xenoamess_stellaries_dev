shishan_code_refresh_vivhite_effect = {
	remove_modifier = shishan_code_vivhite_base
	remove_modifier = shishan_code_vivhite_engineering_step
	remove_modifier = shishan_code_vivhite_society_step
	if = {
		limit = { any_owned_leader = { has_leader_flag = shishan_code_vivhite } }
		add_modifier = { modifier = shishan_code_vivhite_base days = -1 }
		add_modifier = {
			modifier = shishan_code_vivhite_engineering_step
			days = -1
			mult = this.shishan_code_maintenance_count
		}
		add_modifier = {
			modifier = shishan_code_vivhite_society_step
			days = -1
			mult = value:shishan_code_social_penalty_count
		}
	}
}
