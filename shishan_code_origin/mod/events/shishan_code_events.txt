			target = from

				remove_trait = leader_trait_shishan_xp_restore_boost
				clear_variable = shishan_code_vivhite_restore_goal
				clear_variable = shishan_code_vivhite_restore_factor

				exile_leader_as = shishan_code_vivhite_backup@root
			}
		}
		if = {
			limit = { exists = event_target:shishan_code_vivhite_backup@root }
			set_country_flag = shishan_code_vivhite_pending_reboot
			country_event = { id = shishan_code.42 }

country_event = {
	id = shishan_code.41
	title = shishan_code.41.name
	desc = shishan_code.41.desc
	picture = GFX_shishan_event
	is_triggered_only = yes
	trigger = { has_country_flag = shishan_code_vivhite_pending_reboot }
	option = {
		name = shishan_code.41.a
		trigger = {
			resource_stockpile_compare = { resource = energy value >= 500 }
			resource_stockpile_compare = { resource = alloys value >= 200 }
			exists = event_target:shishan_code_vivhite_backup@root
		}
		add_resource = { energy = -500 alloys = -200 }
		clone_leader = {
			target = event_target:shishan_code_vivhite_backup@root
			tier = leader_tier_legendary
		}
		last_created_leader = {
			set_leader_flag = shishan_code_vivhite
			change_leader_portrait = shishan_code_vivhite
			save_event_target_as = shishan_code_vivhite@root
		}
		# add_experience is affected by the new leader's experience gain bonuses.
		# Probe the current multiplier before restoring current-level experience.
		if = {
			limit = { check_variable = { which = shishan_code_vivhite_backup_xp value > 0 } }
			last_created_leader = {
				add_experience = 1
				root = {
					set_variable = {
						which = shishan_code_vivhite_xp_gain_factor
						value = prev.trigger:has_experience
					}
				}
			}
			if = {
				limit = { check_variable = { which = shishan_code_vivhite_xp_gain_factor value > 0 } }
				divide_variable = {
					which = shishan_code_vivhite_backup_xp
					value = shishan_code_vivhite_xp_gain_factor
				}
				change_variable = { which = shishan_code_vivhite_backup_xp value = -1 }
				last_created_leader = { add_experience = root.shishan_code_vivhite_backup_xp }
			}
			else = {
				last_created_leader = {
					set_variable = { which = shishan_code_vivhite_restore_goal value = root.shishan_code_vivhite_backup_xp }
					add_trait = leader_trait_shishan_xp_restore_boost
					leader_event = { id = shishan_code.43 days = 1 }
				}
			}
		}
		event_target:shishan_code_vivhite_backup@root = {
			remove_leader_flag = shishan_code_vivhite
		}
		kill_exiled_leader = shishan_code_vivhite_backup@root
		clear_global_event_target = shishan_code_vivhite_backup@root
		remove_country_flag = shishan_code_vivhite_pending_reboot
		clear_variable = shishan_code_vivhite_backup_xp
		clear_variable = shishan_code_vivhite_xp_gain_factor
		shishan_code_refresh_vivhite_effect = yes
	}
	option = { name = shishan_code.41.b }
}
